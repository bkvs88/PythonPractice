### Take an integer n and print its multiplication table from 1 to 20.
num_inp = int(input("enter any integer number that need table from 1 to 20"))
print(f"Displaying {num_inp} table ")
for i in range(1,21):
    val=num_inp*i
    print(f"{num_inp} X {i} = {val}")


### Calculate the sum of numbers from 1 to n using a loop.
num_inp = int(input("enter any number that need sum from 1 to n"))
i =1
sum=0
while i <=num_inp :
    sum = sum+i
    i=i+1

print(f"Total sum from 1 to {i} is {sum}")

### Calculate the factorial of a number without using any built-in factorial function.
num_inp = int(input("enter any number for factorial"))
fact=1
temp=num_inp
while num_inp >= 1 :
    fact = fact * num_inp
    num_inp = num_inp - 1

print(f"Total factorial of {temp} is {fact}")

### Count how many vowels exist in a user-provided string.
inp_string=input("Enter string to check number of vowels")
print (f"string entered is {inp_string}")
inp_string=inp_string.lower()
vowels = "aeiou"
vowel_count=0
for char in inp_string:
    # print(char)
    if char in vowels:
        vowel_count=vowel_count+1
print(f'number of vowels found in ""{inp_string}"" are {vowel_count}')
